                if (isCtrlOrCmd && isShift && (key === 'c' || keyCode === 67)) {

                }

                // Ctrl+Shift+K (Firefox Web Console) - keyCode 75
                if (isCtrlOrCmd && isShift && (key === 'k' || keyCode === 75)) {
                    return true;
                }

                // Ctrl+Shift+E (Firefox Network) - keyCode 69
                if (isCtrlOrCmd && isShift && (key === 'e' || keyCode === 69)) {
                    return true;
                }

                // Ctrl+U / Cmd+Opt+U (View Source) - keyCode 85
                if ((isCtrlOrCmd && (key === 'u' || keyCode === 85)) ||
                    (e.metaKey && isAlt && (key === 'u' || keyCode === 85))) {
                    return true;
                }

                // Ctrl+S / Cmd+S (Save Page) - keyCode 83
                if (isCtrlOrCmd && (key === 's' || keyCode === 83)) {

        (function () {
            const canvas = document.getElementById('snow-canvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');

            let width = canvas.width = window.innerWidth;
            let height = canvas.height = window.innerHeight;

            const particleCount = 85;
            const particles = [];

            for (let i = 0; i < particleCount; i++) {
                const isPetal = Math.random() < 0.35;
                particles.push({
                    x: Math.random() * width,
                    y: Math.random() * height,
                    radius: isPetal ? (Math.random() * 2.8 + 1.2) : (Math.random() * 2.0 + 0.8),
                    isPetal: isPetal,
                    rotation: Math.random() * Math.PI * 2,
                    rotationSpeed: (Math.random() - 0.5) * 0.04,
                    speedY: isPetal ? (Math.random() * 1.1 + 0.5) : (Math.random() * 0.8 + 0.4),
                    speedX: Math.random() * 0.7 - 0.35,
                    opacity: Math.random() * 0.65 + 0.3,
                    swing: Math.random() * 2 * Math.PI,
                    swingSpeed: Math.random() * 0.025 + 0.012
                });
            }

            function drawSnow() {
                ctx.clearRect(0, 0, width, height);

                for (let i = 0; i < particleCount; i++) {
                    const p = particles[i];

                    ctx.save();
                    if (p.isPetal) {
                        ctx.translate(p.x, p.y);
                        ctx.rotate(p.rotation);
                        ctx.beginPath();
                        ctx.ellipse(0, 0, p.radius * 1.5, p.radius * 0.75, 0, 0, Math.PI * 2);
                        ctx.fillStyle = `rgba(255, 42, 68, ${p.opacity})`;
                        ctx.shadowColor = 'rgba(255, 42, 68, 0.75)';
                        ctx.shadowBlur = 6;
                        ctx.fill();
                        p.rotation += p.rotationSpeed;
                    } else {
                        ctx.beginPath();
                        ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2, false);
                        ctx.fillStyle = `rgba(255, 255, 255, ${p.opacity})`;
                        ctx.shadowColor = 'rgba(255, 255, 255, 0.45)';
                        ctx.shadowBlur = p.radius > 2 ? 4 : 0;
                        ctx.fill();
                    }
                    ctx.restore();

                    // Movement update
                    p.swing += p.swingSpeed;
                    p.y += p.speedY;
                    p.x += Math.sin(p.swing) * 0.7 + p.speedX;

                    // Wrap around boundaries
                    if (p.y > height + 10) {
                        p.y = -10;
                        p.x = Math.random() * width;
                    }
                    if (p.x > width + 10) {
                        p.x = -10;
                    } else if (p.x < -10) {
                        p.x = width + 10;
                    }
                }

                requestAnimationFrame(drawSnow);
            }

            window.addEventListener('resize', () => {
                width = canvas.width = window.innerWidth;
                height = canvas.height = window.innerHeight;
            });

            drawSnow();
        })();

        async function sha256Hex(message) {
            try {
                const msgBuffer = new TextEncoder().encode(message);
                const hashBuffer = await crypto.subtle.digest('SHA-256', msgBuffer);
                const hashArray = Array.from(new Uint8Array(hashBuffer));
                return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
            } catch (e) {
                return '';
            }
        }

        // Utility to check if two member lists are structurally identical (prevents unnecessary re-renders)
        function isRosterEqual(a, b) {
            if (!Array.isArray(a) || !Array.isArray(b)) return false;
            if (a.length !== b.length) return false;
            for (let i = 0; i < a.length; i++) {
                const itemA = a[i];
                const itemB = b[i];
                if (!itemA || !itemB) return false;
                if (itemA.id !== itemB.id ||
                    itemA.name !== itemB.name ||
                    itemA.role !== itemB.role ||
                    itemA.img !== itemB.img ||
                    itemA.fb !== itemB.fb ||
                    itemA.ig !== itemB.ig) {
                    return false;
                }
            }
            return true;
        }

            const listEl = document.getElementById('admin-live-list');

            if (!listEl) return;
            if (countEl) countEl.innerText = `[${liveMembers.length}]`;

            if (liveMembers.length === 0) {
                if (!listEl.querySelector('.admin-empty-msg')) {
                    listEl.innerHTML = '<div class="admin-empty-msg" style="text-align: center; padding: 25px; color: rgba(255,255,255,0.4); font-size: 0.8rem; font-family: \'Orbitron\', sans-serif;">NO MEMBERS YET</div>';
                }
                return;
            } else {
                const emptyMsg = listEl.querySelector('.admin-empty-msg');
                if (emptyMsg) emptyMsg.remove();
            }

            const memberIds = new Set(liveMembers.map(m => m.id));
            Array.from(listEl.children).forEach(child => {
                const mid = child.getAttribute('data-admin-id');
                if (mid && !memberIds.has(mid)) {
                    child.remove();
                }
            });

            liveMembers.forEach((m, index) => {
                let item = listEl.querySelector(`:scope > [data-admin-id="${m.id}"]`);
                const isEditing = editingMemberId === m.id;
                const itemClass = 'admin-live-item' + (isEditing ? ' is-editing' : '');
                const displayImg = (m.img && m.img.trim() !== '') ? m.img : 'logo.png';
                const roleDisplay = ROLE_NAMES[m.role] || (m.role || 'MEMBER').toUpperCase();

                const itemHTML = `
                    <div class="admin-live-item-clickable" onclick="editMember('${m.id}')">
                        <img src="${displayImg}" alt="${escapeHtml(m.name)}" onerror="this.src='logo.png'">
                        <div class="admin-live-meta">
                            <strong>${escapeHtml(m.name)}</strong>
                            <span>${roleDisplay}</span>
                        </div>
                    </div>
                    <div class="admin-live-actions">
                        <button type="button" class="btn-edit-item" onclick="editMember('${m.id}')" title="แก้ไข">
                            <i class="fas fa-pen"></i>
                        </button>
                        <button type="button" class="btn-remove" onclick="deleteMember('${m.id}')" title="ลบ">
                            <i class="fas fa-trash-can"></i>
                        </button>
                    </div>
                `;

                if (!item) {
                    item = document.createElement('div');
                    item.className = itemClass;
                    item.id = `admin-item-${m.id}`;
                    item.setAttribute('data-admin-id', m.id);
                    item.innerHTML = itemHTML;
                    if (index < listEl.children.length) {
                        listEl.insertBefore(item, listEl.children[index]);
                    } else {
                        listEl.appendChild(item);
                    }
                } else {
                    if (item.className !== itemClass) item.className = itemClass;
                    if (item.innerHTML.trim() !== itemHTML.trim()) {
                        item.innerHTML = itemHTML;
                    }
                    if (listEl.children[index] !== item) {
                        listEl.insertBefore(item, listEl.children[index] || null);
                    }
                }
            });
        }

        function isAdminLoggedIn() {
            if (adminSessionMemory) return true;
            try {
                return sessionStorage.getItem(ADMIN_SESSION_KEY) === '1';
            } catch (e) {
                return false;
            }
        }

        function syncAdminScreen() {
            const login = document.getElementById('admin-login');
            const manage = document.getElementById('admin-manage');
            const membersView = document.getElementById('view-members');
            if (!login || !manage) return;

            if (isAdminLoggedIn()) {
                login.classList.add('hidden-view');
                manage.classList.remove('hidden-view');
                if (membersView) membersView.classList.add('hidden-view');
                refreshRosterViews();
            } else {
                manage.classList.add('hidden-view');
                login.classList.remove('hidden-view');
            }
        }

        async function handleAdminLogin(event) {
            event.preventDefault();
            const email = (document.getElementById('admin-email').value || '').trim().toLowerCase();
            const password = (document.getElementById('admin-password').value || '').trim();
            const errorEl = document.getElementById('admin-login-error');

            try {
                const token = await sha256Hex(email + '::' + _AUTH_KEY + '::' + password);

                if (token === _AUTH_VAULT || token === _AUTH_VAULT_ALT) {
                    adminSessionMemory = true;
                    try {
                        sessionStorage.setItem(ADMIN_SESSION_KEY, '1');
                    } catch (e) { }
                    if (errorEl) errorEl.textContent = '';
                    document.getElementById('admin-email').value = '';
                    document.getElementById('admin-password').value = '';
                    syncAdminScreen();
                    history.replaceState(null, '', '#admin');
                    showRosterToast('เข้าสู่ระบบสำเร็จ ยินดีต้อนรับ! 👑');
                    return false;
                }
            } catch (err) { }

            if (errorEl) errorEl.textContent = 'อีเมลหรือรหัสผ่านไม่ถูกต้อง';
            return false;
        }

        function adminLogout() {
            adminSessionMemory = false;
            try {
                sessionStorage.removeItem(ADMIN_SESSION_KEY);
            } catch (e) { }
            syncAdminScreen();
            navigateTo('members');
            showRosterToast('ออกจากระบบเรียบร้อย');
        }

        function previewAdminImage(input) {
            const file = input.files && input.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function (event) {
                const img = new Image();
                img.onload = function () {
                    const canvas = document.createElement('canvas');
                    const maxDim = 360;
                    let width = img.width;
                    let height = img.height;

                    if (width > height) {
                        if (width > maxDim) {
                            height = Math.round((height * maxDim) / width);
                            width = maxDim;
                        }
                    } else {
                        if (height > maxDim) {
                            width = Math.round((width * maxDim) / height);
                            height = maxDim;
                        }
                    }

                    canvas.width = width;
                    canvas.height = height;
                    const ctx = canvas.getContext('2d');
                    ctx.drawImage(img, 0, 0, width, height);

                    pendingImageData = canvas.toDataURL('image/jpeg', 0.82);
                    document.getElementById('add-img-preview').src = pendingImageData;
                    document.getElementById('add-img').value = '';
                };
                img.src = event.target.result;
            };
            reader.readAsDataURL(file);
        }

        // Add auto preview for direct URL input
        document.addEventListener('input', function(e) {
            if (e.target && e.target.id === 'add-img' && e.target.value.trim()) {
                pendingImageData = '';
                document.getElementById('add-img-preview').src = e.target.value.trim();
            }
        });

        function handleAddMember(event) {
            event.preventDefault();
            const name = (document.getElementById('add-name').value || '').trim();
            if (!name) return false;

            const imgInput = (document.getElementById('add-img').value || '').trim();
            const displayImg = pendingImageData || imgInput || 'logo.png';
            const fb = (document.getElementById('add-fb').value || '').trim();
            const ig = (document.getElementById('add-ig').value || '').trim();
            const role = (document.getElementById('add-role').value || 'founder').toLowerCase();

            if (editingMemberId) {
                const idx = liveMembers.findIndex(m => m.id === editingMemberId);
                if (idx !== -1) {
                    liveMembers[idx] = {
                        id: editingMemberId,
                        name: name,
                        role: role,
                        img: displayImg,
                        fb: fb,
                        ig: ig
                    };
                    showRosterToast('อัปเดตข้อมูลสำเร็จ! ✨');
                }
                cancelEditMember();
            } else {
                const newMember = {
                    id: 'm-' + Date.now(),
                    name: name,
                    role: role,
                    img: displayImg,
                    fb: fb,
                    ig: ig
                };
                liveMembers.unshift(newMember);
                showRosterToast('เพิ่มสมาชิกเรียบร้อย! 🎉');
                document.getElementById('admin-add-form').reset();
                document.getElementById('add-img-preview').src = 'logo.png';
                pendingImageData = '';
                setAdminRole('founder');
            }

            saveRoster();
            refreshRosterViews();
            return false;
        }

        function editMember(id) {
            const member = liveMembers.find(m => m.id === id);
            if (!member) return;
            editingMemberId = id;

            document.getElementById('add-name').value = member.name || '';
            document.getElementById('add-img').value = (member.img && !member.img.startsWith('data:')) ? member.img : '';
            document.getElementById('add-img-preview').src = member.img || 'logo.png';
            document.getElementById('add-fb').value = member.fb || '';
            document.getElementById('add-ig').value = member.ig || '';
            pendingImageData = (member.img && member.img.startsWith('data:')) ? member.img : '';

            setAdminRole(member.role || 'member');

            document.getElementById('admin-form-heading').innerHTML = '<i class="fas fa-user-pen" style="margin-right: 6px; color: #ff2a44;"></i> EDIT MEMBER';
            document.getElementById('btn-admin-submit').innerText = 'SAVE CHANGES →';
            document.getElementById('btn-admin-cancel').classList.remove('hidden-view');

            const itemEl = document.getElementById(`admin-item-${id}`);
            if (itemEl) {
                document.querySelectorAll('.admin-live-item').forEach(el => el.classList.remove('is-editing'));
                itemEl.classList.add('is-editing');
            }

            document.getElementById('admin-add-form').scrollIntoView({ behavior: 'smooth' });
        }

        function cancelEditMember() {
            editingMemberId = null;
            document.getElementById('admin-add-form').reset();
            document.getElementById('add-img-preview').src = 'logo.png';
            pendingImageData = '';
            setAdminRole('founder');
            document.getElementById('admin-form-heading').innerHTML = '<i class="fas fa-user-plus" style="margin-right: 6px; color: #fff;"></i> ADD MEMBER';
            document.getElementById('btn-admin-submit').innerText = 'ADD NAME →';
            document.getElementById('btn-admin-cancel').classList.add('hidden-view');
            document.querySelectorAll('.admin-live-item').forEach(el => el.classList.remove('is-editing'));
        }

        async function deleteMember(id) {
            const member = liveMembers.find(m => m.id === id);
            const memberName = member ? member.name : '';
            const confirmed = await showCustomConfirm('ต้องการลบรายชื่อนี้ใช่หรือไม่?', memberName);
            if (!confirmed) return;

            const itemEl = document.getElementById(`admin-item-${id}`);
            if (itemEl) {
                itemEl.classList.add('is-removing');
            }
            setTimeout(() => {
                liveMembers = liveMembers.filter(m => m.id !== id);
                if (editingMemberId === id) {
                    cancelEditMember();
                }
                saveRoster();
                refreshRosterViews();
                showRosterToast('ลบรายชื่อเรียบร้อย! 🗑️');
            }, 200);
        }

        /* --- ðŸ’¿ ULTRA-RESPONSIVE LOCAL AUDIO MUSIC ENGINE --- */
        /* ================================================= */
        const audio = document.getElementById('bg-audio') || new Audio('music.mp3');
        audio.src = 'music.mp3';
        audio.loop = true;
        audio.preload = 'auto';

        let currentVolume = 80;
        let isMuted = false;
        let isPlaying = false;
        let userPausedManually = false;

        audio.volume = currentVolume / 100;

        function updatePlayerUI(playing) {
            isPlaying = playing;
            const container = document.getElementById('player-container');
            const playIcon = document.getElementById('play-icon');
            const musicStatus = document.getElementById('music-status');

            if (playing) {
                container.classList.add('playing');
                playIcon.className = 'fas fa-pause vinyl-action-icon';
                if (musicStatus) musicStatus.innerText = 'NOW PLAYING';
            } else {
                container.classList.remove('playing');
                playIcon.className = 'fas fa-play vinyl-action-icon';
                if (musicStatus) musicStatus.innerText = 'PAUSED';
            }
        }

        function forcePlayAudio() {
            if (userPausedManually) return;
            if (audio.paused) {
                const playPromise = audio.play();
                if (playPromise !== undefined) {
                    playPromise.then(() => {
                        updatePlayerUI(true);
                        removeGestureListeners();
                    }).catch(e => {
                        // Will be unlocked on first touch/mouse movement
                    });
                }
            }
        }

        function togglePlay() {
            if (audio.paused) {
                userPausedManually = false;
                audio.play().then(() => {
                    updatePlayerUI(true);
                    removeGestureListeners();
                }).catch(e => {
                    console.log("Audio play error:", e);
                });
            } else {
                userPausedManually = true;
                removeGestureListeners();
                audio.pause();
                updatePlayerUI(false);
            }
        }

        function changeVolume(val) {
            currentVolume = parseInt(val, 10);
            if (currentVolume === 0) {
                audio.muted = true;
                isMuted = true;
                updateVolumeIcon(0);
            } else {
                if (isMuted) {
                    audio.muted = false;
                    isMuted = false;
                }
                audio.volume = currentVolume / 100;
                updateVolumeIcon(currentVolume);
            }
            document.getElementById('vol-text').innerText = currentVolume + '%';
        }

        function toggleMute(e) {
            if (e) e.stopPropagation();
            if (isMuted) {
                audio.muted = false;
                isMuted = false;
                const restoreVol = currentVolume > 0 ? currentVolume : 80;
                currentVolume = restoreVol;
                audio.volume = restoreVol / 100;
                document.getElementById('vol-slider').value = restoreVol;
                document.getElementById('vol-text').innerText = restoreVol + '%';
                updateVolumeIcon(restoreVol);
            } else {
                audio.muted = true;
                isMuted = true;
                document.getElementById('vol-slider').value = 0;
                document.getElementById('vol-text').innerText = '0%';
                updateVolumeIcon(0);
            }
        }

        function updateVolumeIcon(vol) {
            const icon = document.getElementById('vol-icon');
            if (!icon) return;
            if (vol === 0) icon.className = 'fas fa-volume-xmark vol-btn-icon';
            else if (vol < 50) icon.className = 'fas fa-volume-low vol-btn-icon';
            else icon.className = 'fas fa-volume-high vol-btn-icon';
        }

        // Attach Click to Vinyl Disc
        document.getElementById('master-play').addEventListener('click', function (e) {
            const infoBox = document.getElementById('vinylInfoBox');
            if (window.innerWidth <= 768 && infoBox) {
                infoBox.classList.toggle('active');
            }
            togglePlay();
        });

        // Audio state listeners
        audio.addEventListener('play', () => updatePlayerUI(true));
        audio.addEventListener('pause', () => updatePlayerUI(false));

        /* --- âš¡ Immediate Autoplay & Gesture Unlock Engine --- */
        const gestureEvents = ['click', 'touchstart', 'touchend', 'mousedown', 'pointerdown', 'mousemove', 'scroll', 'wheel', 'keydown'];

        function onUserGesture() {
            forcePlayAudio();
        }

        function addGestureListeners() {
            gestureEvents.forEach(evt => {
                window.addEventListener(evt, onUserGesture, { passive: true });
                document.addEventListener(evt, onUserGesture, { passive: true });
            });
        }

        function removeGestureListeners() {
            gestureEvents.forEach(evt => {
                window.removeEventListener(evt, onUserGesture);
                document.removeEventListener(evt, onUserGesture);
            });
        }

        // 1. Try playing immediately as script loads
        forcePlayAudio();

        // 2. Add full gesture listeners for instantaneous unlock
        addGestureListeners();

        // 3. When DOM is ready and window is loaded, retry immediately
        document.addEventListener('DOMContentLoaded', () => {
            loadRoster();
            startRealtimeStream();
            forcePlayAudio();
            if (window.location.hash === '#admin') {
                navigateTo('admin');
            } else if (window.location.hash === '#members') {
                navigateTo('members');
            }
        });

        window.addEventListener('load', () => {
            forcePlayAudio();
        });

        // 4. When page becomes visible/focused
        document.addEventListener('visibilitychange', () => {
            if (document.visibilityState === 'visible') {
                if (!userPausedManually && audio.paused) {
                    forcePlayAudio();
                }
            }
        });
